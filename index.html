Searching / Sorting

// From: CarbonCopy (CC) - Efficient Array Searching (Big-O Log N)
public int binarySearch(int[] dataSet, int targetId) {
    int left = 0, right = dataSet.length - 1;
    while (left <= right) {
        int mid = left + (right - left) / 2;
        if (dataSet[mid] == targetId) return mid;
        if (dataSet[mid] < targetId) left = mid + 1;
        else right = mid - 1;
    }
    return -1;
}

// From: ChooseYourOwnAdventure - Path Branching Logic
public int calculatePathDepth(Node currentPath) {
    if (currentPath.isEndNode()) {
        return 1; // Base case
    }
    return 1 + calculatePathDepth(currentPath.getNextNode());
}
